% Block diagram of accelerator_alexnet_v2_int16 (ROWS=8, COLS=32).
% Same visual language as img/abstract_arch.png: black outlines, white fill,
% nested boxes, double-headed arrows, sans-serif labels.
% Build:  pdflatex alexnet_v2_arch.tex   (from img/)
\documentclass[tikz,border=4pt]{standalone}
\usepackage[T1]{fontenc}
\usepackage{helvet}
\renewcommand{\familydefault}{\sfdefault}
\usetikzlibrary{arrows.meta,calc}

\begin{document}
\begin{tikzpicture}[
  font=\small,
  line width=0.9pt,
  blk/.style={draw, fill=white, align=center, inner sep=3pt},
  pe/.style={blk, minimum width=1.05cm, minimum height=0.68cm, font=\footnotesize, inner sep=1pt},
  rf/.style={blk, minimum width=0.40cm, minimum height=0.68cm, inner sep=0pt},
  bus/.style={{Latex[length=2.8mm]}-{Latex[length=2.8mm]}, line width=1.3pt},
  flow/.style={-{Latex[length=1.7mm]}, line width=0.6pt},
  lbl/.style={font=\footnotesize, align=center},
]

% ---------------- PE array ----------------
\def\cpitch{2.0}   % column pitch (cm)
\def\rpitch{0.98}  % row pitch (cm)

% cells: drawn columns 0,1,2 | ... | 31  (index 3 is the ellipsis)
\foreach \r in {0,...,7} {
  \foreach \c in {0,...,4} {
    \pgfmathsetmacro{\cx}{\c*\cpitch}
    \pgfmathsetmacro{\cy}{-\r*\rpitch}
    \ifnum\c=3
      \node[font=\large] at (\cx+0.72,\cy) {$\cdots$};
    \else
      \node[rf, anchor=west] (rf\r\c) at (\cx,\cy) {\rotatebox{90}{\scriptsize RF}};
      \node[pe, anchor=west] (pe\r\c) at (rf\r\c.east) {PE};
    \fi
  }
}

% activation flow (left -> right)
\foreach \r in {0,...,7} {
  \draw[flow] (pe\r0.east) -- (rf\r1.west);
  \draw[flow] (pe\r1.east) -- (rf\r2.west);
  \draw[flow] (pe\r2.east) -- ++(0.38,0);
  \draw[flow] ($(rf\r4.west)+(-0.38,0)$) -- (rf\r4.west);
}
% partial-sum flow (top -> bottom)
\foreach \r [evaluate=\r as \rn using int(\r+1)] in {0,...,6} {
  \foreach \c in {0,1,2,4} {
    \draw[flow] (pe\r\c.south) -- (pe\rn\c.north);
  }
}

% array frame + title + foot note
\coordinate (aSW) at ($(rf70.west)+(-0.6,-1.05)$);
\coordinate (aNE) at ($(pe04.east)+(0.6,0)$);
\coordinate (aNE) at ($(aNE|-pe04.north)+(0,1.25)$);
\draw[line width=1.1pt] (aSW) rectangle (aNE);
\node[anchor=north, lbl] at ($(aNE)!0.5!(aNE-|aSW)+(0,-0.12)$)
  {\textbf{PE array $8\times32$}\\
   256$\times$ \texttt{pe\_tiny\_xpu\_int16} (INT16 MAC, 48-bit accumulator)};
\node[anchor=south, lbl] at ($(aSW)!0.5!(aSW-|aNE)+(0,0.12)$)
  {RF: 3$\times$ \texttt{mem\_bsg\_1r1w\_8x8} per PE (768 in total)};

% ---------------- Memories on the left ----------------
\path let \p1=(aSW), \p2=(aNE) in
  node[blk, minimum width=1.7cm, minimum height={\y2-\y1}, anchor=south east]
  (gbuf) at ($(aSW)+(-1.6,0)$)
  {\rotatebox{90}{\parbox{8cm}{\centering
     \textbf{Global buffer}\\[3pt]
     \texttt{mem\_openeye\_sp\_128x256}\\
     $128\times256$\,bit, single port}}};
\draw[bus] (gbuf.east) -- (gbuf.east-|aSW);

\path let \p1=(aSW), \p2=(aNE) in
  node[blk, dashed, minimum width=1.7cm, minimum height={\y2-\y1}, anchor=south east]
  (dram) at ($(gbuf.south west)+(-1.9,0)$)
  {\rotatebox{90}{\parbox{8cm}{\centering
     \textbf{External DRAM}\\[3pt]
     \footnotesize excluded from synthesis}}};
\draw[bus] (dram.east) -- (gbuf.west)
  node[midway, above, lbl, yshift=2pt] {512-bit link};

% ---------------- Outer frame ----------------
\coordinate (oSW) at ($(dram.south west)+(-0.4,-0.4)$);
\coordinate (oNE) at ($(aNE)+(0.4,0.8)$);
\draw[line width=1.3pt] (oSW) rectangle (oNE);
\node[anchor=north west, lbl] at ($(oSW|-oNE)+(0.15,-0.1)$)
  {\texttt{accelerator\_alexnet\_v2\_int16} (\texttt{ROWS}=8, \texttt{COLS}=32)};

\end{tikzpicture}
\end{document}
